Dialog Example

Confirm Action

Are you sure you want to proceed with this action?

 
<button commandfor="confirm-dialog" command="show-modal">
  Open Dialog
</button>
<dialog id="confirm-dialog">
  <!-- Dialog content -->
  <button commandfor="confirm-dialog" command="close" value="cancel">
    Cancel
  </button>
  <button commandfor="confirm-dialog" command="close" value="confirm">
    Confirm
  </button>
</dialog>

Custom Command Example

<button commandfor="demo-image" command="--rotate-landscape">
  Landscape
</button>
<button commandfor="demo-image" command="--rotate-portrait">
  Portrait
</button>
<img id="demo-image" src="photo.jpg">

<script>
  const image = document.getElementById("demo-image");
  image.addEventListener("command", (event) => {
    if (event.command == "--rotate-landscape") {
      image.style.rotate = "-90deg"
    } else if (event.command == "--rotate-portrait") {
      image.style.rotate = "0deg"
    }
  });
</script>
